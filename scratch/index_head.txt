    function analyzeAndClassifyMedia(mediaList) {
      let loadedCount = 0;
      const targetCount = mediaList.length;
      let completed = false;

      const safetyTimer = setTimeout(() => {
        if (!completed) { completed = true; console.warn('??꾩븘????媛뺤젣 湲곕룞'); startMediaRotators(); }
      }, 3500);

      mediaList.forEach(item => {
        let finalUrl = item.url;
        if (finalUrl.includes('drive.google.com') || finalUrl.includes('docs.google.com')) {
          const id = extractGoogleDriveFileId(finalUrl);
          if (id) {
            finalUrl = item.type === 'video' 
              ? `https://docs.google.com/uc?export=download&id=${id}`
              : `https://drive.google.com/thumbnail?sz=w1280&id=${id}`;
            item.url = finalUrl;
          }
        }

        // isVertical???대? ?뺤쓽??寃쎌슦 利됱떆 遺꾨쪟
        if (typeof item.isVertical === 'boolean') { sortItem(item); return; }

        if (item.type === 'video') {
          const v = document.createElement('video');
          v.referrerPolicy = 'no-referrer';
          v.src = finalUrl;
          v.addEventListener('loadedmetadata', () => { if (!completed) { item.isVertical = v.videoHeight > v.videoWidth; sortItem(item); } });
          v.addEventListener('error', () => { if (!completed) { loadedCount++; checkComplete(); } });
        } else {
          const img = new Image();
          img.referrerPolicy = 'no-referrer';
          img.src = finalUrl;
          img.onload  = () => { if (!completed) { item.isVertical = img.naturalHeight > img.naturalWidth; sortItem(item); } };
          img.onerror = () => { if (!completed) { loadedCount++; checkComplete(); } };
        }
      });

      function sortItem(item) {
        if (item.isVertical) verticalList.push(item); else horizontalList.push(item);
        loadedCount++;
        checkComplete();
      }
      function checkComplete() {
        if (loadedCount >= targetCount && !completed) { clearTimeout(safetyTimer); completed = true; startMediaRotators(); }
      }
    }

    function processMediaList(list) {

      const shuffled = shuffleArray([...list]);
      const verticals   = shuffled.filter(m => m.isVertical);
      const hVideos     = shuffled.filter(m => !m.isVertical && m.type === 'video');
      const hImages     = shuffled.filter(m => !m.isVertical && m.type !== 'video');
      analyzeAndClassifyMedia([...verticals, ...hVideos, ...hImages]);
    }

    function toProxyUrl(item) {
      const id   = extractGoogleDriveFileId(item.url);
      if (id) {
        let finalUrl = item.type === 'video' 
          ? `https://docs.google.com/uc?export=download&id=${id}`
          : `https://drive.google.com/thumbnail?sz=w1280&id=${id}`;
        
        const isTokenValid = googleAccessToken && googleTokenExpiry && Date.now() < parseInt(googleTokenExpiry);
        if (isTokenValid) {
          finalUrl += `&access_token=${googleAccessToken}`;
        }
        return finalUrl;
      }
      return item.url;
    }

    function shuffleArray(array) {
      for (let i = array.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [array[i], array[j]] = [array[j], array[i]];
      }
      return array;
    }

    function convertSheetValuesToObjects(values) {
      if (!values || values.length === 0) return [];
      const headers = values[0].map(h => h.trim());
      const result = [];
      for (let i = 1; i < values.length; i++) {

        const obj = {};
        headers.forEach((header, index) => {
          obj[header] = row[index] !== undefined ? row[index].trim() : '';
        });
        result.push(obj);
      }
      return result;
    }

    function parseCSV(csvText) {
      const lines = []; let row = ['']; let inQ = false;
      for (let i = 0; i < csvText.length; i++) {
        const c = csvText[i], n = csvText[i+1];
        if (c === '"') { if (inQ && n === '"') { row[row.length-1] += '"'; i++; } else inQ = !inQ; }
        else if (c === ',' && !inQ) row.push('');
        else if ((c === '\r' || c === '\n') && !inQ) { if (c === '\r' && n === '\n') i++; lines.push(row); row = ['']; }
        else row[row.length-1] += c;
      }
      if (row.length > 1 || row[0] !== '') lines.push(row);
      if (lines.length === 0) return [];
      const headers = lines[0].map(h => h.trim());
      const result = [];
      for (let i = 1; i < lines.length; i++) {
        const l = lines[i];
        if (l.length === 1 && l[0] === '') continue;
        const obj = {};
        headers.forEach((h,j) => obj[h] = l[j] ? l[j].trim() : '');
        result.push(obj);
      }
      return result;
    }

    function toggleControlPanel() {
      const panel = document.querySelector('.control-panel');
      const toggleIcon = document.getElementById('panel-toggle-icon');
      if (!panel) return;

      if (panel.classList.contains('show')) {
        panel.classList.remove('show');
